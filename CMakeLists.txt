cmake_minimum_required(VERSION 3.20)
project(ShiftNest VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libmicrohttpd libpq libsodium libcjson)
find_package(Threads REQUIRED)
add_library(optimizer backend/optimizer.c)
target_include_directories(optimizer PUBLIC backend)
target_compile_options(optimizer PRIVATE -Wall -Wextra -Wpedantic)
add_executable(shiftnest backend/main.c backend/db.c)
target_link_libraries(shiftnest PRIVATE optimizer PkgConfig::DEPS Threads::Threads m)
target_compile_options(shiftnest PRIVATE -Wall -Wextra -Wpedantic)
add_executable(optimizer_tests tests/optimizer_test.c)
target_link_libraries(optimizer_tests PRIVATE optimizer m)
if(SANITIZE AND NOT WIN32)
  foreach(t optimizer shiftnest optimizer_tests)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME optimizer COMMAND optimizer_tests)
